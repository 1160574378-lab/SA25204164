易知对于连续型分布Rayleigh分布： \[F(x)=1-e^{-\frac{x^{2} }{2\sigma ^{2} } } (x\ge 0)\]

\[F^{- 1} (u)=\sqrt{-2\sigma ^{2}\ln_{}{(1-u)}}\]

采用逆变换法生成10000个服从Rayleigh分布的样本，自定义3个函数函数，分别是rayleigh_inv <- function(n, sigma),validate_rayleigh <- function(sigma_values, n_samples = 10000, method = “inv”),plot_validation <- function(results)。 取多个不同大于0的\(\sigma=0.5,1.0,2.0,5.0\)。并验证生成样本的众数是否接近理论众数\(\sigma\)(密度函数求导），通过相对误差来反映，最后绘制对应的验证直方图。具体详见代码解释。结果如下：

# 使用逆变换抽样法生成瑞利分布随机数
rayleigh_inv <- function(n, sigma) {
  # 瑞利分布的累积分布函数：F(x) = 1 - exp(-x^2/(2*sigma^2))
  # 逆函数：F^{-1}(u) = sigma * sqrt(-2 * log(1 - u))
  u <- runif(n)
  x <- sigma * sqrt(-2 * log(1 - u))
  return(x)
}


# 主函数：生成样本并验证众数
validate_rayleigh <- function(sigma_values, n_samples = 10000, method = "inv") {
  
  results <- list()
  
  for (sigma in sigma_values) {
    # 生成样本
    if (method == "inv") {
      samples <- rayleigh_inv(n_samples, sigma)
    } else if (method == "polar") {
      samples <- rayleigh_polar(n_samples, sigma)
    } else {
      samples <- rayleigh_builtin(n_samples, sigma)
    }
    
    # 计算样本众数（使用核密度估计的峰值）
    dens <- density(samples)
    empirical_mode <- dens$x[which.max(dens$y)]
    
    # 理论众数
    theoretical_mode <- sigma
    
    # 计算相对误差
    relative_error <- abs(empirical_mode - theoretical_mode) / theoretical_mode * 100
    
    # 存储结果
    results[[as.character(sigma)]] <- list(
      sigma = sigma,
      samples = samples,
      empirical_mode = empirical_mode,
      theoretical_mode = theoretical_mode,
      relative_error = relative_error,
      density = dens
    )
    
    # 打印结果
    cat(sprintf("σ = %.2f: 理论众数 = %.2f, 经验众数 = %.2f, 相对误差 = %.2f%%\n",
                sigma, theoretical_mode, empirical_mode, relative_error))
  }
  
  return(results)
}

首先编写函数beta_ar_sampler <- function(n, a, b, M = NULL)， 目的是利用接受-拒绝法的思想在Beta(a,b)分布生成大小为n的随机样本。取 g(x)=1，M是目标分布密度 f(x) 的上界(不小于)，确保 M ≥ f(x) 对所有 0<x<1 成立。根据比值决定是否接受候选样本。具体详见代码解释。

对于Beta(3,2)，众数在 x = 0.5，最大密度约为1.875。生成大小为1000的随机样本。输出接受率，样本均值和方差，绘制样本的直方图，并叠加理论密度曲线。具体详见代码解释。结果如下：

# 定义接受-拒绝法采样函数
beta_ar_sampler <- function(n, a, b, M = NULL) {
  "
  使用接受-拒绝法从贝塔分布生成随机样本
  
  参数:
  n: 样本大小
  a, b: 贝塔分布的参数
  M: 接受-拒绝法中的常数，如果为NULL则自动计算
  
  返回:
  list: 包含样本和接受率的列表
  "
  
  # 如果没有提供M，则计算合适的M值
  if (is.null(M)) {
    # 贝塔分布的众数出现在 x = (a-1)/(a+b-2)
    if (a > 1 && b > 1) {
      mode <- (a - 1) / (a + b - 2)
      max_density <- dbeta(mode, a, b)
    } else {
      # 如果参数<=1，众数在边界，我们取一个安全的上界
      max_density <- dbeta(0.5, a, b) * 1.5
    }
    M <- max_density * 1.1  # 加10%的安全边际
  }
  
  samples <- numeric(n)
  trials <- 0
  accepted <- 0
  
  while (accepted < n) {
    trials <- trials + 1
    
    # 从建议分布（均匀分布）生成候选样本
    x_candidate <- runif(1, 0, 1)
    
    # 从均匀分布生成比较值
    u <- runif(1, 0, 1)
    
    # 计算目标密度与建议密度的比值
    # 建议分布是U(0,1)，密度为1
    target_density <- dbeta(x_candidate, a, b)
    ratio <- target_density / M
    
    # 接受-拒绝准则
    if (u <= ratio) {
      accepted <- accepted + 1
      samples[accepted] <- x_candidate
    }
  }
  
  acceptance_rate <- n / trials
  
  return(list(samples = samples, acceptance_rate = acceptance_rate))
}

# 定义生成混合正态分布的函数
generate_mixture_normal <- function(n, p1, mu1 = 0, mu2 = 3, sigma1 = 1, sigma2 = 1) {
  # 生成混合指标
  mixture_indicator <- rbinom(n, 1, p1)
  
  # 根据混合指标生成样本
  sample_data <- numeric(n)
  sample_data[mixture_indicator == 1] <- rnorm(sum(mixture_indicator == 1), mu1, sigma1)
  sample_data[mixture_indicator == 0] <- rnorm(sum(mixture_indicator == 0), mu2, sigma2)
  
  return(sample_data)
}

# 定义混合分布的密度函数
mixture_density <- function(x, p1, mu1 = 0, mu2 = 3, sigma1 = 1, sigma2 = 1) {
  p1 * dnorm(x, mu1, sigma1) + (1 - p1) * dnorm(x, mu2, sigma2)
}

# 定义蒙特卡洛模拟函数
monte_carlo_t_test <- function(distribution, n, n_sim = 10000, alpha = 0.05) {
  # n: 样本大小
  # n_sim: 模拟次数
  # alpha: 显著性水平
  
  type_I_errors <- numeric(n_sim)
  
  for (i in 1:n_sim) {
    # 根据指定的分布生成样本
    if (distribution == "chisq") {
      sample_data <- rchisq(n, df = 1)  # χ²(1)，均值=1
      mu0 <- 1
    } else if (distribution == "uniform") {
      sample_data <- runif(n, 0, 2)     # Uniform(0,2)，均值=1
      mu0 <- 1
    } else if (distribution == "exponential") {
      sample_data <- rexp(n, rate = 1)  # Exponential(1)，均值=1
      mu0 <- 1
    }
    
    # 执行单样本t检验
    t_test_result <- t.test(sample_data, mu = mu0, alternative = "two.sided")
    
    # 记录是否犯Type I错误（p值 < α）
    type_I_errors[i] <- t_test_result$p.value < alpha
  }
  
  # 计算经验Type I错误率
  empirical_alpha <- mean(type_I_errors)
  
  return(empirical_alpha)
}

              dimnames = list(NULL,
                              c("Bonf_rej","BH_rej","Bonf_TP","BH_TP")))

for(r in 1:R){
  # 生成 p 值
  p0 <- runif(m0)                    # U(0,1)
  p1 <- rbeta(m1, 0.1, 1)            # Beta(0.1,1)  stochastically smaller
  p  <- c(p0, p1)
  H  <- c(rep(0L,m0), rep(1L,m1))    # 真实状态 0=原假设 1=备择
  
  # 校正
  bonf_p  <- p.adjust(p, method = "bonferroni")
  bh_p    <- p.adjust(p, method = "BH")
  
  # 拒绝指标
  bonf_rej <- bonf_p <= alpha
  bh_rej   <- bh_p   <= alpha
  
  res[r, "Bonf_rej"] <- sum(bonf_rej)
  res[r, "BH_rej"]   <- sum(bh_rej)
  res[r, "Bonf_TP"]  <- sum(bonf_rej & H)   # 真正阳性
  res[r, "BH_TP"]    <- sum(bh_rej   & H)
}

题目要求

实现 Cramér-von Mises 双样本检验的置换检验，使用公式 (10.14)，并应用到 Example 10.1 和 10.2 的数据。

Cramér-von Mises 距离定义为：

\[\omega^2 = \int \int (F(x) - G(y))^2 dH(x, y)\]

其中 \(H(x, y)\) 是 \(X\) 和 \(Y\) 的联合累积分布函数。

基于秩的检验统计量：

对于等分布检验，相应的检验统计量基于联合经验分布，因此是数据秩的函数。

步骤：

  1. 计算 \(X\) 样本的秩 \(r_i\)，\(i = 1, \ldots, n\)
  2. 计算 \(Y\) 样本的秩 \(s_j\)，\(j = 1, \ldots, m\)
  3. 计算中间统计量 \(U\)：

\[U = n\sum_{i=1}^{n}(r_i - i)^2 + m\sum_{j=1}^{m}(s_j - j)^2\]

  1. 计算 Cramér-von Mises 双样本统计量（公式 10.14）：

\[W^2 = \frac{U}{nm(n+m)} - \frac{4mn-1}{6(m+n)}\]

注意： \(W^2\) 的大值表示显著差异。

置换检验框架：

  1. 计算观测统计量 \(W^2_{\text{obs}}\)
  2. 进行 \(R\) 次（如 999 次）随机置换：
    • 合并两组数据
    • 随机重新分配到两组
    • 计算置换统计量 \(W^2_{\text{perm}}\)
  3. 计算 p 值：

\[p\text{-value} = \frac{\#\{W^2_{\text{perm}} \geq W^2_{\text{obs}}\} + 1}{R + 1}\]

应用数据：

  • Example 10.1 和 10.2: chickwts 数据集
    • 比较 soybean vs linseed 两种饲料对小鸡体重的影响
    • 检验两组体重分布是否相同
# Exercise 10.3: Cramér-von Mises 双样本置换检验

# 定义 Cramér-von Mises 统计量（基于公式 10.14）
cramer_von_mises <- function(x, y) {
  n <- length(x)
  m <- length(y)
  N <- n + m
  
  # 合并数据并计算秩
  z <- c(x, y)
  ranks <- rank(z)
  
  # X 样本的秩
  r <- ranks[1:n]
  # Y 样本的秩
  s <- ranks[(n+1):N]
  
  # 计算 U
  U <- n * sum((r - (1:n))^2) + m * sum((s - (1:m))^2)
  
  # 计算 W^2（公式 10.14）
  W2 <- U / (n * m * N) - (4 * m * n - 1) / (6 * N)
  
  return(W2)
}

# 置换检验函数
permutation_test_cvm <- function(x, y, R = 999) {
  n <- length(x)
  m <- length(y)
  z <- c(x, y)
  
  # 观测统计量
  W2_obs <- cramer_von_mises(x, y)
  
  # 置换
  W2_perm <- numeric(R)
  for (i in 1:R) {
    perm_idx <- sample(n + m)
    x_perm <- z[perm_idx[1:n]]
    y_perm <- z[perm_idx[(n+1):(n+m)]]
    W2_perm[i] <- cramer_von_mises(x_perm, y_perm)
  }
  
  # 计算 p 值
  p_value <- mean(c(W2_obs, W2_perm) >= W2_obs)
  
  return(list(statistic = W2_obs, p.value = p_value))
}

# Count 5 统计量函数（适用于不等样本量）
count5_stat <- function(x_list) {
  k <- length(x_list)
  n <- sapply(x_list, length)
  
  # 中心化每个样本
  x_centered <- lapply(x_list, function(x) x - mean(x))
  
  # 合并所有数据
  all_data <- unlist(x_centered)
  
  # 找出每组的极端点数量
  counts <- numeric(k)
  for (i in 1:k) {
    # 计算该组有多少点是所有数据中的极端点
    x_i <- x_centered[[i]]
    counts[i] <- sum(abs(x_i) >= quantile(abs(all_data), 0.95))
  }
  
  # 返回最大极端点数
  return(max(counts))
}

# 置换检验函数
count5_permutation_test <- function(x_list, R = 999) {
  k <- length(x_list)
  n <- sapply(x_list, length)
  
  # 观测统计量
  obs_stat <- count5_stat(x_list)
  
  # 合并所有数据（中心化后）
  x_centered <- lapply(x_list, function(x) x - mean(x))
  pooled <- unlist(x_centered)
  N <- length(pooled)
  
  # 置换复制
  perm_stats <- numeric(R)
  for (i in 1:R) {
    # 随机置换
    perm_data <- sample(pooled)
    
    # 按原样本量分组
    x_perm <- list()
    start <- 1
    for (j in 1:k) {
      x_perm[[j]] <- perm_data[start:(start + n[j] - 1)]
      start <- start + n[j]
    }
    
    perm_stats[i] <- count5_stat(x_perm)
  }
  
  # 计算 p 值
  p_value <- mean(c(obs_stat, perm_stats) >= obs_stat)
  
  return(list(statistic = obs_stat, p.value = p_value))
}

题目：证明在连续情形下，采用提议分布 \(g(r|s)\) 的 Metropolis–Hastings (M-H) 采样器的平稳分布（stationary distribution） 是目标分布 \(f(x)\) 。

即需要证明：如果马尔可夫链是由M-H算法生成的，那么其平稳分布为目标密度 （只需在连续情形下说明）。

证明：

M–H算法定义（连续情形）

  1. 假设当前状态为 \(X_t = s\)

  2. 从提议分布 \(g(r|s)\) 生成候选点 \(r\)

  3. 计算接受概率： \[ \alpha(s, r) = \min\left\{1, \frac{f(r) g(s|r)}{f(s) g(r|s)} \right\} \]

  4. 以概率 \(\alpha(s,r)\) 接受 \(r\)，否则保持原状态。

转移核（Transition Kernel）为： \[ K(s, r) = g(r|s) \alpha(s, r) + \left[ 1 - \int g(u|s) \alpha(s, u)\,du \right] \delta_s(r) \] 其中 \(\delta_s(r)\) 表示拒绝时链保持在原状态。

平稳分布定义： 若概率密度 \(f(x)\) 满足：

\[ \int f(s) \, K(s, r) \, ds = f(r) \]

则称 \(f(x)\) 是该马尔可夫链的 平稳分布（stationary distribution）.

Detailed Balance 条件: 平稳分布的充分条件之一是 详细平衡条件：

\[ f(s) \, K(s, r) = f(r) \, K(r, s) \]

若此式成立，链的平稳分布就是 \(f(x)\)。

证明 M-H 满足详细平衡: 当 \(r \neq s\) 时：

转移密度： \[ K(s, r) = g(r|s) \, \alpha(s, r) \] \[ K(r, s) = g(s|r) \, \alpha(r, s) \] 接受概率： \[ \alpha(s, r) = \min\left\{1, \frac{f(r) g(s|r)}{f(s) g(r|s)} \right\} \] \[ \alpha(r, s) = \min\left\{1, \frac{f(s) g(r|s)}{f(r) g(s|r)} \right\} \] 情况 1： 若
\[ \frac{f(r)g(s|r)}{f(s)g(r|s)} \le 1 \]
则
\[ \alpha(s,r) = \frac{f(r)g(s|r)}{f(s)g(r|s)}, \quad \alpha(r,s) = 1 \]

带入可得： \[ f(s)K(s,r) = f(s)g(r|s)\frac{f(r)g(s|r)}{f(s)g(r|s)} = f(r)g(s|r) = f(r)K(r,s) \]

情况 2： 若
\[ \frac{f(r)g(s|r)}{f(s)g(r|s)} > 1 \]
则
\[ \alpha(s,r) = 1, \quad \alpha(r,s) = \frac{f(s)g(r|s)}{f(r)g(s|r)} \]

同样可得： \[ f(s)K(s,r) = f(s)g(r|s) = f(r)K(r,s) \]

当 \(r = s\) 时，总概率守恒条件保证等式成立。

因此 详细平衡在 M-H 算法中成立。 由于 M-H 算法满足详细平衡条件：

\[ f(s)K(s, r) = f(r)K(r, s) \]

对 \(r\) 积分得到： \[ \int f(s)K(s, r)\, ds = f(r) \]

因此 \(f(x)\) 为 M-H 链的平稳分布。经过足够迭代后，样本分布收敛于目标分布 \(f(x)\)。

# 目标分布（标准拉普拉斯）的对数密度函数
logtarget <- function(x) {
  return(-abs(x) - log(2))  # 标准拉普拉斯密度取log
}

# Metropolis采样器
metropolis <- function(n_iter, sigma, x0=0) {
  samples <- numeric(n_iter)
  samples[1] <- x0
  accepted <- 0
  
  for (t in 2:n_iter) {
    # 提议：从 N(x_{t-1}, sigma^2) 采样
    proposal <- rnorm(1, mean = samples[t-1], sd = sigma)
    
    # 计算接受概率（对数）
    log_alpha <- logtarget(proposal) - logtarget(samples[t-1])
    
    # 接受一步
    if (log(runif(1)) < log_alpha) {
      samples[t] <- proposal
      accepted <- accepted + 1
    } else {
      samples[t] <- samples[t-1]
    }
  }
  accept_rate <- accepted / (n_iter - 1)
  list(samples = samples, accept_rate = accept_rate)
}

# Gibbs 采样器
gibbs_sampler <- function(n_iter, n, a, b, x0 = 0, y0 = 0.5) {
  # 存储样本
  X <- numeric(n_iter)
  Y <- numeric(n_iter)
  
  # 初始值
  X[1] <- x0
  Y[1] <- y0
  
  # Gibbs 采样迭代
  for (i in 2:n_iter) {
    # 从条件分布 X|Y ~ Binomial(n, y) 采样
    X[i] <- rbinom(1, size = n, prob = Y[i-1])
    
    # 从条件分布 Y|X ~ Beta(x+a, n-x+b) 采样
    Y[i] <- rbeta(1, shape1 = X[i] + a, shape2 = n - X[i] + b)
  }
  
  return(data.frame(X = X, Y = Y))
}

  n_iter <- batch
  repeat {
    chains <- do.call(chain_func, c(list(n_iter = n_iter), param_list))
    
    if (is.list(chains)) {
      # 处理多个参数，如Gibbs例子
      rhat_x <- gelman.diag(chains$X)$psrf[, "Point est."] 
      rhat_y <- gelman.diag(chains$Y)$psrf[, "Point est."]
      cat(sprintf("迭代 %d 次: X Gelman-Rubin R̂ = %.3f, Y Gelman-Rubin R̂ = %.3f\n", 
                  n_iter, rhat_x, rhat_y))
      converged <- all(rhat_x < target_rhat) & all(rhat_y < target_rhat)
    } else {
      # 单参数链，如Metropolis例子
      rhat <- gelman.diag(chains)$psrf[, "Point est."]
      cat(sprintf("迭代 %d 次: Gelman-Rubin R̂ = %.3f\n", n_iter, rhat))
      converged <- all(rhat < target_rhat)
    }
    
    if (converged || n_iter >= max_iter) break
    n_iter <- n_iter + batch
  }
  return(chains)
}

11.4

参考Example 11.3中的贝叶斯预测应用，证明在几何生存模型Geometric(p)中，派生参数 \(\psi(p)=\frac{p}{1-p}\)不依赖于个体已达到的年龄（这在一般模型中不成立）。

解答：

理论证明

几何分布的无记忆性质

在几何生存模型中，个体在第 \(k\) 个时期存活的概率为： \[P_T(T = k) = p^k(1-p), \quad k = 0, 1, 2, \ldots\]

关键性质：几何分布具有无记忆性（memoryless property） \[P(T > s + t \mid T > s) = P(T > t)\]

证明：

对于已存活 \(s\) 个时期的个体，条件概率为： \[P(T > s + t \mid T > s) = \frac{P(T > s + t)}{P(T > s)} = \frac{p^{s+t}}{p^s} = p^t = P(T > t)\]

因此，条件分布与初始分布相同： \[P(T = s + k \mid T > s) = P(T = k) = p^k(1-p)\]

条件期望剩余寿命为： \[E[T - s \mid T > s] = \sum_{k=0}^{\infty} k \cdot p^k(1-p) = \frac{p}{1-p} = \psi(p)\]

这与初始期望寿命 \(E[T] = \frac{p}{1-p}\) 完全相同，证明了 \(\psi(p)\) 不依赖于已达到的年龄。

课上题目

EM算法：区间删失指数分布参数估计

问题

设 \(X_1, \ldots, X_n \stackrel{\text{i.i.d}}{\sim} \text{Exp}(\lambda)\)，概率密度函数为： \[f(x|\lambda) = \lambda e^{-\lambda x}, \quad x \geq 0\]

但我们只能观测到区间数据 \((U_i, V_i)\)，其中 \(X_i \in [U_i, V_i)\)，这是一种区间删失（interval censoring）问题。

解答

1. 完全数据的对数似然函数

如果能观测到完整数据 \(X_1, \ldots, X_n\)，对数似然函数为： \[\ell_c(\lambda) = \sum_{i=1}^n \log f(x_i|\lambda) = n\log\lambda - \lambda\sum_{i=1}^n x_i\]

2. 观测数据的似然函数

对于区间删失数据，观测到 \(X_i \in [U_i, V_i)\) 的概率为： \[P(U_i \leq X_i < V_i) = \int_{U_i}^{V_i} \lambda e^{-\lambda x} dx = e^{-\lambda U_i} - e^{-\lambda V_i}\]

观测数据的对数似然函数为： \[\ell(\lambda) = \sum_{i=1}^n \log\left(e^{-\lambda U_i} - e^{-\lambda V_i}\right)\]

3. EM算法推导

E步（Expectation Step）：

计算完全数据对数似然的条件期望： \[Q(\lambda|\lambda^{(t)}) = E_{X|U,V,\lambda^{(t)}}[\ell_c(\lambda)]\]

对于指数分布的区间删失，给定 \(X_i \in [U_i, V_i)\)，条件分布为截断指数分布： \[E[X_i | U_i \leq X_i < V_i, \lambda^{(t)}] = U_i + \frac{1}{\lambda^{(t)}} - \frac{V_i e^{-\lambda^{(t)}(V_i - U_i)}}{1 - e^{-\lambda^{(t)}(V_i - U_i)}}\]

简化记为： \[\bar{X}_i^{(t)} = U_i + \frac{1 - (1 + \lambda^{(t)}(V_i - U_i))e^{-\lambda^{(t)}(V_i - U_i)}}{\lambda^{(t)}(1 - e^{-\lambda^{(t)}(V_i - U_i)})}\]

则： \[Q(\lambda|\lambda^{(t)}) = n\log\lambda - \lambda\sum_{i=1}^n \bar{X}_i^{(t)}\]

M步（Maximization Step）：

对 \(Q(\lambda|\lambda^{(t)})\) 关于 \(\lambda\) 求导并令其为0： \[\frac{\partial Q}{\partial \lambda} = \frac{n}{\lambda} - \sum_{i=1}^n \bar{X}_i^{(t)} = 0\]

得到更新公式： \[\lambda^{(t+1)} = \frac{n}{\sum_{i=1}^n \bar{X}_i^{(t)}}\]

em_exponential <- function(U, V, lambda_init = 1, max_iter = 1000, tol = 1e-8) {
  lambda <- lambda_init
  lambda_history <- numeric(max_iter)
  
  for (iter in 1:max_iter) {
    # E步：计算条件期望
    delta <- V - U
    exp_term <- exp(-lambda * delta)
    
    # 条件期望 E[X_i | U_i <= X_i < V_i]
    X_bar <- U + (1 - (1 + lambda * delta) * exp_term) / (lambda * (1 - exp_term))
    
    # M步：更新lambda
    lambda_new <- n / sum(X_bar)
    
    # 记录历史
    lambda_history[iter] <- lambda_new
    
    # 检查收敛
    if (abs(lambda_new - lambda) < tol) {
      cat("EM算法在第", iter, "次迭代后收敛\n")
      return(list(
        lambda = lambda_new,
        iterations = iter,
        history = lambda_history[1:iter]
      ))
    }
    
    lambda <- lambda_new
  }
  
  warning("EM算法未在最大迭代次数内收敛")
  return(list(
    lambda = lambda,
    iterations = max_iter,
    history = lambda_history
  ))
}

理论证明

(1) EM算法收敛性证明

定理：在正则条件下，EM算法产生的序列 \(\{\lambda^{(t)}\}\) 单调增加观测数据的似然函数值，且收敛到局部极大值点。

证明要点： 1. 单调性：由EM算法的性质，\(\ell(\lambda^{(t+1)}) \geq \ell(\lambda^{(t)})\) 2. 有界性：似然函数在参数空间内有上界 3. 收敛性：由单调有界定理，序列收敛

(2) 收敛到MLE的证明

对于指数分布的区间删失问题： - 似然函数是凹函数（在对数尺度下） - 存在唯一的全局最大值点 - EM算法的不动点即为MLE

(3) 线性收敛速度

EM算法的收敛速度由缺失信息比例（fraction of missing information）决定： \[\text{收敛率} \approx 1 - \frac{I_{\text{obs}}(\lambda)}{I_{\text{complete}}(\lambda)}\]

其中：

  • \(I_{\text{complete}}(\lambda) = n/\lambda^2\)（完全数据的Fisher信息）
  • \(I_{\text{obs}}(\lambda)\)（观测数据的Fisher信息）

对于区间删失，缺失信息比例较大，因此收敛速度为线性收敛。

  )
}

# 快速 table 函数（针对整数向量）
table_fast <- function(x, y) {
  # 输入验证
  if (!is.integer(x)) x <- as.integer(x)
  if (!is.integer(y)) y <- as.integer(y)
  if (anyNA(x) || anyNA(y)) {
    stop("输入不能包含缺失值")
  }
  if (length(x) != length(y)) {
    stop("两个向量长度必须相同")
  }
  
  # 获取唯一值
  x_levels <- sort(unique(x))
  y_levels <- sort(unique(y))
  
  # 创建空矩阵
  result <- matrix(0, 
                   nrow = length(x_levels), 
                   ncol = length(y_levels),
                   dimnames = list(x_levels, y_levels))
  
  # 填充计数
  for (i in seq_along(x)) {
    row_idx <- match(x[i], x_levels)
    col_idx <- match(y[i], y_levels)
    result[row_idx, col_idx] <- result[row_idx, col_idx] + 1
  }
  
  result
}

# 使用快速 table 改进卡方检验

chisq_test_faster <- function(x, y) {
  # 使用快速 table
  observed <- table_fast(x, y)
  
  # 计算期望频数
  row_sums <- rowSums(observed)
  col_sums <- colSums(observed)
  n <- sum(observed)
  expected <- outer(row_sums, col_sums) / n
  
  # 计算卡方统计量
  chi_squared <- sum((observed - expected)^2 / expected)
  df <- (nrow(observed) - 1) * (ncol(observed) - 1)
  p_value <- pchisq(chi_squared, df, lower.tail = FALSE)
  
  list(
    statistic = chi_squared,
    parameter = df,
    p.value = p_value
  )
}

解答： 这个问题是关于遗传连锁 (Genetic Linkage) 的模型。我们需要估计参数 \(\theta\) 的后验分布。为了实现 Gibbs 采样，引入一个潜在变量（Latent Variable）将多项分布进行扩充。

观测数据：\(y = (125, 18, 20, 34)\)。 将第一组 \(y_1=125\) 分为两部分：\(z_{1a}\) (对应概率 1/2) 和 \(z_{1b}\) (对应概率 \(\theta/4\))。

两个条件分布用于迭代：

  • 更新 \(z_{1b}\) (给定 \(\theta\)):服从二项分布：\(z_{1b} | \theta, y \sim \text{Binomial}\left(125, \frac{\theta}{2+\theta}\right)\)
  • 更新 \(\theta\) (给定 \(z_{1b}\)):假设先验分布为 Uniform (即 Beta(1,1))，则后验为 Beta 分布：\(\theta | z_{1b}, y \sim \text{Beta}(z_{1b} + 34 + 1, 18 + 20 + 1) = \text{Beta}(z_{1b} + 35, 39)\)

cppFunction('
  NumericVector gibbs_rcpp(int N, int thin) {
    NumericVector mat(N);
    double x = 0.5; // 初始值 theta
    double y = 0;   // 潜在变量 z_1b
    int n_sample = 125; 
    
    // Beta分布参数常数部分
    double shape2 = 39.0; // 18 + 20 + 1
    
    for(int i = 0; i < N; i++) {
      for(int j = 0; j < thin; j++) {
        // 1. 更新 latent variable z_1b
        // R::rbinom(n, p)
        y = R::rbinom(n_sample, x / (2.0 + x));
        
        // 2. 更新 theta
        // R::rbeta(shape1, shape2)
        x = R::rbeta(y + 35.0, shape2);
      }
      mat[i] = x;
    }
    return(mat);
  }
')

gibbs_r <- function(N, thin) {
  mat <- numeric(N)
  x <- 0.5
  y <- 0
  n_sample <- 125
  shape2 <- 39
  
  for(i in 1:N) {
    for(j in 1:thin) {
      # 1. 更新 latent variable
      y <- rbinom(1, n_sample, x / (2 + x))
      # 2. 更新 theta
      x <- rbeta(1, y + 35, shape2)
    }
    mat[i] <- x
  }
  return(mat)
}
